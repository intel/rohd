Returns the FLC JSON for the sole source occurrence with definitionName, or null if no traces exist for that module.

Throws an ArgumentError when more than one source occurrence requested definitionName. Use flcForModuleInstance in that case so a caller cannot accidentally cross-probe the wrong occurrence.

Map<String, Object>? flcForModule(String definitionName) {
  final targets = <Module>[];
  void walk(Module m) {
    if (m.definitionName == definitionName) {
      targets.add(m);
    }
    m.subModules.forEach(walk);
  }

  walk(module);

  if (targets.isEmpty) {
    return null;
  }
  if (targets.length > 1) {
    throw ArgumentError.value(
      definitionName,
      'definitionName',
      'matches ${targets.length} source module occurrences; use '
          'flcForModuleInstance instead',
    );
  }

  return flcForModuleInstance(targets.single);
}
