void flushBlock()

Flushes buffered value changes to disk as a VcData block.

After flushing, the changes are cleared from memory and the block is recorded in blockIndex for later read-back. This enables incremental writing where only recent unflushed changes remain in memory while historical data lives on disk.

Does nothing if no changes are buffered.

Implementation

void flushBlock() {
  if (_changes.isEmpty) {
    return;
  }
  if (!_headerWritten) {
    throw StateError('Must call writeHeader() before flushing blocks');
  }

  // Keep equal-time updates to each signal in emission order. They may
  // otherwise be reversed by List.sort, changing the final value in a frame.
  _changes.sort((a, b) {
    final cmp = a.time.compareTo(b.time);
    if (cmp != 0) {
      return cmp;
    }
    final handleCmp = a.handleIndex.compareTo(b.handleIndex);
    return handleCmp != 0 ? handleCmp : a.sequence.compareTo(b.sequence);
  });

  final blockStart = _changes.first.time;
  final blockEnd = _changes.last.time;

  // A frame is the carry-over state before any changes at the block's
  // first timestamp. Those changes must remain in the encoded chains so
  // standard FST readers can observe them.
  final frameValues = List<String>.from(_nextFrameBase);

  final blockOffset = _file.positionSync();
  _writeVcDataBlock(
    blockStartTime: blockStart,
    blockEndTime: blockEnd,
    frameValues: frameValues,
  );
  final blockEndPos = _file.positionSync();

  // Record block in the index for read-back
  _blockIndex.add(
    FstBlockIndex(
      fileOffset: blockOffset,
      sectionLength: blockEndPos - blockOffset - 1,
      startTime: blockStart,
      endTime: blockEnd,
    ),
  );
  _vcSectionCount++;

  // Update global time range
  if (_vcSectionCount == 1) {
    _startTime = blockStart;
  }
  _endTime = blockEnd;

  // Carry-over state for next block's frame
  _nextFrameBase = List<String>.from(_currentValues);
  _changes.clear();
}
