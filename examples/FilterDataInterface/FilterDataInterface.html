FilterDataInterface({int dataWidth = 16}) : _dataWidth = dataWidth {
  setPorts(
    [Logic.port('sampleIn', dataWidth), Logic.port('validIn')],
    [FilterPortTag.inputPorts],
  );

  setPorts(
    [Logic.port('dataOut', dataWidth), Logic.port('validOut')],
    [FilterPortTag.outputPorts],
  );
}
